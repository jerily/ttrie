cmake_minimum_required(VERSION 3.18)
project(ttrie VERSION 0.1.1 LANGUAGES C)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
option(TTRIE_BUILD_TCL "Build the Tcl extension" ON)
option(BUILD_SHARED_LIBS "Build a loadable Tcl extension" ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# The C library is always static and has no Tcl dependency.
add_library(ttrie_core STATIC src/ttrie.c)
add_library(ttrie::core ALIAS ttrie_core)
set_target_properties(ttrie_core PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME core)
target_include_directories(ttrie_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/ttrie>)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ttrie_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS ttrie_core EXPORT ttrieTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES src/ttrie.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/ttrie)
install(EXPORT ttrieTargets NAMESPACE ttrie:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ttrie)
configure_package_config_file(cmake/ttrieConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/ttrieConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ttrie)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/ttrieConfigVersion.cmake
    COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ttrieConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/ttrieConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ttrie)

if(TTRIE_BUILD_TCL)
    find_path(TCL_INCLUDE_PATH NAMES tcl.h HINTS ${TCL_INCLUDE_DIR}
        PATH_SUFFIXES tcl9.1 tcl9.0 tcl8.6 tcl REQUIRED)
    file(STRINGS "${TCL_INCLUDE_PATH}/tcl.h" TCL_VERSION_LINE REGEX "^#[ \t]*define TCL_VERSION[ \t]+\"")
    string(REGEX MATCH "[0-9]+\\.[0-9]+" TTRIE_TCL_VERSION "${TCL_VERSION_LINE}")
    if(TTRIE_TCL_VERSION VERSION_LESS 8.6 OR NOT TTRIE_TCL_VERSION VERSION_LESS 10)
        message(FATAL_ERROR "Tcl 8.6 or 9.x is required")
    endif()
    if(TTRIE_TCL_VERSION VERSION_LESS 9)
        set(TTRIE_TCL_REQUIREMENT "8.6-9")
    else()
        set(TTRIE_TCL_REQUIREMENT "9.0-10")
    endif()
    find_program(TCL_TCLSH NAMES tclsh${TTRIE_TCL_VERSION} HINTS "${TCL_INCLUDE_PATH}/../bin")
    add_library(ttrie src/ttrie_tcl.c)
    target_include_directories(ttrie PRIVATE ${TCL_INCLUDE_PATH})
    target_compile_definitions(ttrie PRIVATE TTRIE_VERSION="${PROJECT_VERSION}")
    target_link_libraries(ttrie PRIVATE ttrie_core)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(ttrie PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(BUILD_SHARED_LIBS)
        string(REPLACE "." "" TTRIE_TCL_COMPACT_VERSION "${TTRIE_TCL_VERSION}")
        find_library(TCL_STUB_LIBRARY NAMES tclstub${TTRIE_TCL_VERSION} tclstub${TTRIE_TCL_COMPACT_VERSION} tclstub
            HINTS ${TCL_LIBRARY_DIR} "${TCL_INCLUDE_PATH}/../lib" REQUIRED)
        target_compile_definitions(ttrie PRIVATE USE_TCL_STUBS)
        target_link_libraries(ttrie PRIVATE ${TCL_STUB_LIBRARY})
        set(TTRIE_LOAD "[list load [file join $dir ${CMAKE_SHARED_LIBRARY_PREFIX}ttrie${CMAKE_SHARED_LIBRARY_SUFFIX}] Ttrie]")
    else()
        target_compile_definitions(ttrie PRIVATE STATIC_BUILD)
        set(TTRIE_LOAD "[list load {} Ttrie]")
    endif()
    configure_file(pkgIndex.tcl.in pkgIndex.tcl @ONLY)
    install(TARGETS ttrie LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/ttrie${PROJECT_VERSION}
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl DESTINATION ${CMAKE_INSTALL_LIBDIR}/ttrie${PROJECT_VERSION})
endif()

if(BUILD_TESTING)
    add_executable(test_core tests/core.c)
    target_link_libraries(test_core PRIVATE ttrie_core)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(test_core PRIVATE -UNDEBUG)
    endif()
    add_test(NAME core COMMAND test_core)
    if(TTRIE_BUILD_TCL)
        if(BUILD_SHARED_LIBS)
            if(NOT TCL_TCLSH)
                message(FATAL_ERROR "Tcl shell is required for tests")
            endif()
            set(TTRIE_TEST_SHELL "${TCL_TCLSH}")
            if(UNIX AND CMAKE_NM)
                add_test(NAME tcl_stubs COMMAND ${CMAKE_COMMAND}
                    -DNM=${CMAKE_NM} -DMODULE=$<TARGET_FILE:ttrie>
                    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/stubs.cmake)
            endif()
        else()
            # Only the embedding test executable needs the full Tcl runtime.
            find_library(TCL_LIBRARY NAMES tcl${TTRIE_TCL_VERSION}
                HINTS ${TCL_LIBRARY_DIR} "${TCL_INCLUDE_PATH}/../lib" REQUIRED)
            add_executable(test_embed tests/embed.c)
            target_include_directories(test_embed PRIVATE ${TCL_INCLUDE_PATH})
            target_link_libraries(test_embed PRIVATE ttrie ttrie_core ${TCL_LIBRARY})
            set(TTRIE_TEST_SHELL "$<TARGET_FILE:test_embed>")
        endif()
        add_test(NAME tcl COMMAND ${TTRIE_TEST_SHELL} ${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl)
        set_tests_properties(tcl PROPERTIES ENVIRONMENT "TCLLIBPATH=${CMAKE_CURRENT_BINARY_DIR}")
    endif()
endif()
